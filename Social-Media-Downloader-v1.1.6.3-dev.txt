// Instagram response capture: independently implemented from documented media shapes.
const igMediaCache = new Map();
const metaCaptureStats={parsedBodies:0,videoRecords:0};
function metaCodeFromId(id) {
    const pk=String(id||'').split('_')[0];
    if(!/^\d+$/.test(pk)) return '';
    try {let n=BigInt(pk),code='';const alphabet='ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789-_';while(n>0n){code=alphabet[Number(n%64n)]+code;n/=64n;}return code;}catch(_){return '';}
}
function igImageKey(url) {
    try { return new URL(url, location.origin).pathname; } catch (_) { return ''; }
}
function inspectInstagramMedia(payload) {
    let root=payload;
    if(typeof root==='string') {
        try { root=JSON.parse(root.replace(/^for\s*\(\s*;;\s*\);\s*/,'')); }
        catch (_) { for(const line of root.split('\n')) { try { inspectInstagramMedia(JSON.parse(line)); } catch (_) {} } return; }
    }
    metaCaptureStats.parsedBodies++;
    const seen=new WeakSet(); let count=0;
    function visit(node,inheritedCode='') {
        if(!node || typeof node!=='object' || seen.has(node) || ++count>100000) return;
        seen.add(node);
        const variants=Array.isArray(node.video_versions) ? node.video_versions : [];
        const urls=variants.filter(v=>v && /^https?:\/\//.test(v.url||'')).map(v=>({url:v.url,width:Number(v.width)||0,height:Number(v.height)||0}));
        if(!urls.length && /^https?:\/\//.test(node.video_url||'')) urls.push({url:node.video_url,width:Number(node.dimensions?.width)||0,height:Number(node.dimensions?.height)||0});
        if(urls.length) {
            let code=node.code||node.shortcode||inheritedCode||'';
            const pk=String(node.pk||node.id||'').split('_')[0];
            if(!code && /^\d+$/.test(pk)) {
                try { let n=BigInt(pk); const alphabet='ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789-_'; while(n>0n) {code=alphabet[Number(n%64n)]+code; n/=64n;} } catch (_) {}
            }
            if(code) {
                const images=[node.display_url,node.thumbnail_src,...(node.image_versions2?.candidates||[]).map(v=>v.url)].filter(Boolean).map(igImageKey).filter(Boolean);
                const previous=igMediaCache.get(code);
                metaCaptureStats.videoRecords++;
                igMediaCache.set(code,{code,urls,images:[...new Set([...(previous?.images||[]),...images])],time:Date.now()});
                if(igMediaCache.size>250) igMediaCache.delete(igMediaCache.keys().next().value);
            }
        }
        for(const [key,value] of Object.entries(node)) {
            const inherited=/^(media|video|video_info|if_not_gated_logged_out)$/.test(key)?node.code||node.shortcode||inheritedCode:'';
            visit(value,inherited);
        }
    }
    visit(root);
}
function scanInstagramScripts() {
    for(const script of document.querySelectorAll('script[type="application/json"],script[data-sjs]')) {
        if(/video_versions|video_url/.test(script.textContent||'')) inspectInstagramMedia(script.textContent);
    }
}
function selectInstagramCached(video,code) {
    const poster=igImageKey(video?.poster||video?.getAttribute?.('poster')||'');
    const matches=poster ? [...igMediaCache.values()].filter(r=>Date.now()-r.time<1800000 && r.images.includes(poster)) : [];
    const record=matches.length===1 ? matches[0] : igMediaCache.get(code);
    if(!record || Date.now()-record.time>=1800000) return null;
    const urls=[...record.urls].sort((a,b)=>(b.width*b.height)-(a.width*a.height));
    return {url:urls[0].url,filename:'Instagram_'+record.code+'.mp4'};
}

// ============================================================
// X API MEDIA CACHE (v1.1.0-dev)
// ============================================================
//
// X 的 GraphQL / API JSON 有時會直接帶 video_info.variants。
// 若找到 video/mp4，就優先使用最高 bitrate 的 MP4；
// 找不到時完全不影響舊版流程，會回到 HLS + FFmpeg。
//

const xApiMediaCache = new Map();

function normalizeXVariant(variant) {
    if (!variant || typeof variant !== 'object') return null;

    const url = variant.url || variant.src || '';
    const contentType =
        variant.content_type ||
        variant.contentType ||
        variant.type ||
        '';

    if (!/^https?:\/\//i.test(url)) return null;

    return {
        url,
        contentType,
        bitrate: Number(variant.bitrate) || 0
    };
}

function cacheXMediaRecord(media, tweetId = '') {
    if (!media || typeof media !== 'object') return;

    const mediaId = String(
        media.id_str ||
        media.id ||
        media.media_key?.match?.(/(\d+)$/)?.[1] ||
        ''
    );

    const variants =
        media.video_info?.variants ||
        media.videoInfo?.variants ||
        [];

    if (!mediaId || !Array.isArray(variants) || !variants.length) return;

    const mp4Variants = variants
        .map(normalizeXVariant)
        .filter(Boolean)
        .filter(item =>
            /video\/mp4/i.test(item.contentType) ||
            /\.mp4(?:\?|$)/i.test(item.url)
        )
        .sort((a, b) => b.bitrate - a.bitrate);

    if (!mp4Variants.length) return;

    const previous = xApiMediaCache.get(mediaId);
    const best = mp4Variants[0];

    if (!previous || best.bitrate >= (previous.best?.bitrate || 0)) {
        xApiMediaCache.set(mediaId, {
            mediaId,
            tweetId: String(tweetId || previous?.tweetId || ''),
            best,
            variants: mp4Variants,
            updatedAt: Date.now()
        });

        console.log(
            '[SMD:X API] cached MP4',
            mediaId,
            `${Math.round(best.bitrate / 1000)} kbps`,
            best.url
        );
    }
}

function scanXApiJSON(root) {
    if (!root || typeof root !== 'object') return;

    const seen = new WeakSet();

    function walk(node, tweetId = '') {
        if (!node || typeof node !== 'object') return;
        if (seen.has(node)) return;
        seen.add(node);

        let localTweetId = tweetId;

        // GraphQL Tweet result 常見結構：rest_id + legacy。
        if (node.rest_id && node.legacy && typeof node.legacy === 'object') {
            localTweetId = String(node.rest_id);
        } else if (node.id_str && (node.full_text || node.extended_entities)) {
            localTweetId = String(node.id_str);
        }

        // 常見 legacy.extended_entities.media / entities.media。
        const mediaLists = [
            node.extended_entities?.media,
            node.entities?.media,
            node.legacy?.extended_entities?.media,
            node.legacy?.entities?.media
        ];

        for (const list of mediaLists) {
            if (!Array.isArray(list)) continue;
            for (const media of list) {
                cacheXMediaRecord(media, localTweetId);
            }
        }

        // 防止 API 結構改版：任何帶 video_info.variants 的物件也嘗試收錄。
        if (node.video_info?.variants || node.videoInfo?.variants) {
            cacheXMediaRecord(node, localTweetId);
        }

        if (Array.isArray(node)) {
            for (const item of node) walk(item, localTweetId);
            return;
        }

        for (const value of Object.values(node)) {
            if (value && typeof value === 'object') {
                walk(value, localTweetId);
            }
        }
    }

    walk(root, '');
}

function inspectXApiPayload(payload) {
    try {
        if (!payload) return;
        if (typeof payload === 'string') {
            if (!payload.includes('video_info') && !payload.includes('videoInfo')) return;
            scanXApiJSON(JSON.parse(payload));
        } else if (typeof payload === 'object') {
            scanXApiJSON(payload);
        }
    } catch (_) {
        // API 回應很多，不讓單一非 JSON / 非預期結構影響頁面。
    }
}

function installXApiInterceptor() {
    const host = location.hostname.toLowerCase();
    if (!(host === 'x.com' || host.endsWith('.x.com') || host.includes('twitter.com'))) {
        return;
    }

    try {
        const page = typeof unsafeWindow !== 'undefined' ? unsafeWindow : window;

        if (!page.__SMD_X_API_FETCH_PATCHED__ && typeof page.fetch === 'function') {
            page.__SMD_X_API_FETCH_PATCHED__ = true;
            const originalFetch = page.fetch;

            page.fetch = function(...args) {
                const promise = originalFetch.apply(this, args);

                promise.then(response => {
                    try {
                        const type = response?.headers?.get?.('content-type') || '';
                        const url = response?.url || '';
                        if (
                            /json/i.test(type) &&
                            (/\/i\/api\//i.test(url) || /graphql/i.test(url) || /api\.x\.com/i.test(url))
                        ) {
                            response.clone().text().then(inspectXApiPayload).catch(() => {});
                        }
                    } catch (_) {}
                }).catch(() => {});

                return promise;
            };

            console.log('[SMD:X API] fetch interceptor installed');
        }

        const XHR = page.XMLHttpRequest;
        if (XHR?.prototype && !XHR.prototype.__SMD_X_API_PATCHED__) {
            XHR.prototype.__SMD_X_API_PATCHED__ = true;
            const originalSend = XHR.prototype.send;

            XHR.prototype.send = function(...args) {
                this.addEventListener('load', () => {
                    try {
                        const url = this.responseURL || '';
                        if (!(/\/i\/api\//i.test(url) || /graphql/i.test(url) || /api\.x\.com/i.test(url))) {
                            return;
                        }

                        if (this.responseType === 'json') {
                            inspectXApiPayload(this.response);
                        } else if (!this.responseType || this.responseType === 'text') {
                            inspectXApiPayload(this.responseText);
                        }
                    } catch (_) {}
                }, { once: true });

                return originalSend.apply(this, args);
            };

            console.log('[SMD:X API] XHR interceptor installed');
        }
    } catch (error) {
        console.warn('[SMD:X API] interceptor unavailable; HLS fallback remains active', error);
    }
}

installXApiInterceptor();


// Facebook SPA removes consumed JSON scripts; capture media delivery at document-start.
const fbApiMediaCache = new Map();
function inspectFacebookAPI(payload) {
    const scan = root => {
        const seen = new WeakSet();
        const walk = node => {
            if (!node || typeof node !== 'object' || seen.has(node)) return;
            seen.add(node);
            const id = String(node.id || node.video_id || '');
            if (/^\d+$/.test(id) && (node.videoDeliveryLegacyFields || node.videoDeliveryResponseFragment || node.playable_url || node.browser_native_hd_url)) {
                const urls = [];
                const shapes = new Set();
                let manifest = '';
                const collect = (value, path = '', depth = 0) => {
                    if (!value || typeof value !== 'object' || depth > 12) return;
                    const childId = String(value.id || value.video_id || '');
                    if (depth > 0 && childId && childId !== id && (value.videoDeliveryLegacyFields || value.videoDeliveryResponseFragment)) return;
                    for (const [key, item] of Object.entries(value)) {
                        const next = path ? path + '.' + key : key;
                        if (/url|manifest|dash|progressive|delivery|representation/i.test(key)) shapes.add(next + ':' + (item === null ? 'null' : Array.isArray(item) ? 'array' : typeof item));
                        const native = /^(browser_native_hd_url|browser_native_sd_url|playable_url_quality_hd|playable_url|progressive_url)$/.test(key);
                        const progressive = /progressive/i.test(path) && /^(url|base_url)$/.test(key);
                        if ((native || progressive) && typeof item === 'string' && /^https?:\/\//i.test(item)) {
                            urls.push({url:item,score:/hd|quality_hd/.test(key)?4:/sd/.test(key)?1:/hd/i.test(value.metadata?.quality||'')?4:/sd/i.test(value.metadata?.quality||'')?1:2,width:Number(value.width||value.metadata?.width)||0,height:Number(value.height||value.metadata?.height)||0});
                        }
                        if (typeof item === 'string' && /manifest/i.test(key) && /<MPD/.test(item)) manifest = item;
                        if (item && typeof item === 'object') collect(item,next,depth+1);
                    }
                };
                for (const key of ['videoDeliveryLegacyFields','videoDeliveryResponseFragment']) if (node[key]) collect(node[key],key);
                collect({playable_url:node.playable_url,playable_url_quality_hd:node.playable_url_quality_hd,browser_native_hd_url:node.browser_native_hd_url,browser_native_sd_url:node.browser_native_sd_url});
                const thumbnails = [node.thumbnailImage?.uri, node.thumbnailImage?.url,
                    node.preferred_thumbnail?.image?.uri, node.preferred_thumbnail?.uri,
                    node.previewImage?.uri].filter(url => typeof url === 'string' && /^https?:\/\//i.test(url));
                const previous=fbApiMediaCache.get(id);
                if(urls.length || manifest || shapes.size) {
                    fbApiMediaCache.set(id,{urls:urls.length?urls:previous?.urls||[],manifest:manifest||previous?.manifest||'',shapes:[...shapes].slice(0,100),thumbnails:[...new Set([...(previous?.thumbnails||[]),...thumbnails])]});
                    if(fbApiMediaCache.size>100)fbApiMediaCache.delete(fbApiMediaCache.keys().next().value);
                }
            }
            for (const value of Object.values(node)) walk(value);
        };
        walk(root);
    };
    if (payload && typeof payload === 'object') return scan(payload);
    if (typeof payload !== 'string') return;
    const cleaned = payload.replace(/^for\s*\(;;\);\s*/, '');
    try { scan(JSON.parse(cleaned)); return; } catch (_) {}
    // GraphQL streaming responses may contain one JSON object per line.
    for (const line of cleaned.split(/\r?\n/)) {
        try { scan(JSON.parse(line.replace(/^for\s*\(;;\);\s*/, ''))); } catch (_) {}
    }
}

function installFacebookAPIInterceptor() {
    if (!/(^|\.)facebook\.com$/.test(location.hostname)) return;
    try {
        const page = typeof unsafeWindow !== 'undefined' ? unsafeWindow : window;
        const relevant = url => /facebook\.com/.test(url || '') && /graphql|api\//i.test(url || '');
        if (!page.__SMD_FB_API_FETCH__ && typeof page.fetch === 'function') {
            page.__SMD_FB_API_FETCH__ = true;
            const original = page.fetch;
            page.fetch = function(...args) {
                const result=original.apply(this,args);
                result.then(response => {
                    if(relevant(response.url)) response.clone().text().then(inspectFacebookAPI).catch(()=>{});
                }).catch(()=>{});
                return result;
            };
        }
        const proto=page.XMLHttpRequest?.prototype;
        if(proto && !proto.__SMD_FB_API_XHR__) {
            proto.__SMD_FB_API_XHR__=true;
            const original=proto.send;
            proto.send=function(...args) {
                this.addEventListener('load',()=>{
                    if(!relevant(this.responseURL))return;
                    try {
                        if(this.responseType==='json')inspectFacebookAPI(this.response);
                        else if(!this.responseType || this.responseType==='text')inspectFacebookAPI(this.responseText);
                    }catch(_){}
                },{once:true});
                return original.apply(this,args);
            };
        }
        // Bootstrap JSON may already exist by the time the userscript runs.
        const consume = root => {
            if(root.nodeType!==1)return;
            const scripts=[...(root.matches?.('script[type="application/json"]')?[root]:[]),...root.querySelectorAll('script[type="application/json"]')];
            for(const script of scripts)inspectFacebookAPI(script.textContent);
        };
        const observer=new MutationObserver(records=>{
            for(const record of records)for(const node of record.addedNodes)consume(node);
        });
        observer.observe(document,{childList:true,subtree:true});
        if(document.documentElement)consume(document.documentElement);
    } catch(error) { console.warn('[SMD:FB] API capture unavailable',error); }
}
installFacebookAPIInterceptor();

const capturedResources = [];
const capturedSet = new Set();

function rememberResource(url, type = '') {
    if (!url || capturedSet.has(url)) return;

    capturedSet.add(url);

    capturedResources.push({
        url,
        type,
        time: Date.now()
    });

    if (capturedResources.length > 5000) {
        const removed = capturedResources.shift();

        if (removed) {
            capturedSet.delete(removed.url);
        }
    }
}

try {
    const observer = new PerformanceObserver(list => {
        for (const entry of list.getEntries()) {
            if (entry?.name) {
                rememberResource(
                    entry.name,
                    entry.initiatorType || ''
                );
            }
        }
    });

    observer.observe({
        type: 'resource',
        buffered: true
    });

} catch (error) {
    console.warn('[SMD] PerformanceObserver error', error);
}

    function stopXDownload(switchHLS = false) {
        const control = xControl;
        if (!control || control.stopping) return;
        if (switchHLS && (control.phase !== 'mp4' || !control.abortCurrent)) return;
        control.stopping = true;
        if (!switchHLS) control.cancelled = true;
        const error = controlError(switchHLS ? 'SWITCH_HLS' : 'CANCELLED');
        try {
            if (control.abortCurrent) control.abortCurrent(error);
        } catch (abortError) {
            // Never start HLS when abort fails.
            control.cancelled = false;
            toast('無法中止目前請求：' + abortError.message);
        } finally {
            control.stopping = false;
        }
    }

    function formatBytes(bytes) {
        const n = Number(bytes) || 0;
        if (n <= 0) return '0 B';
        const units = ['B', 'KB', 'MB', 'GB'];
        const i = Math.min(Math.floor(Math.log(n) / Math.log(1024)), units.length - 1);
        const value = n / Math.pow(1024, i);
        return `${value >= 100 || i === 0 ? value.toFixed(0) : value.toFixed(1)} ${units[i]}`;
    }

    function cleanFilename(text) {

    function absoluteURL(url, base = location.href) {
        if (!url) return '';

        try {
            return new URL(url, base).href;
        } catch (_) {
            return '';
        }
    }


    function decodeURL(text) {
        if (!text) return '';

        let result = text;

        for (let i = 0; i < 4; i++) {
            result = result
                .replace(/\\u0026/gi, '&')
                .replace(/\\u003d/gi, '=')
                .replace(/\\u002f/gi, '/')
                .replace(/\\u0025/gi, '%')
                .replace(/\\u003a/gi, ':')
                .replace(/\\\//g, '/')
                .replace(/&amp;/gi, '&');
        }

        return result;
    }

    function isUsableMedia(el) {
        if (!el || !el.isConnected) {
            return false;
        }

        if (
            el.tagName !== 'IMG' &&
            el.tagName !== 'VIDEO'
        ) {
            return false;
        }

        const rect = el.getBoundingClientRect();

        if (
            rect.width < 170 ||
            rect.height < 130
        ) {
            return false;
        }

        const css = getComputedStyle(el);

        if (
            css.display === 'none' ||
            css.visibility === 'hidden' ||
            Number(css.opacity) === 0
        ) {
            return false;
        }

        return true;
    }


    function findMediaFromPoint(x, y) {
        const elements =
            document.elementsFromPoint(x, y);

        for (const el of elements) {
            if (
                (
                    el.tagName === 'IMG' ||
                    el.tagName === 'VIDEO'
                ) &&
                isUsableMedia(el)
            ) {
                return el;
            }
        }


        for (const top of elements.slice(0, 8)) {
            let node = top;

            for (
                let depth = 0;
                depth < 6 && node;
                depth++
            ) {
                if (
                    (
                        node.tagName === 'IMG' ||
                        node.tagName === 'VIDEO'
                    ) &&
                    isUsableMedia(node)
                ) {
                    return node;
                }


                for (
                    const video
                    of node.querySelectorAll?.('video') || []
                ) {
                    if (isUsableMedia(video)) {
                        return video;
                    }
                }


                for (
                    const img
                    of node.querySelectorAll?.('img') || []
                ) {
                    if (isUsableMedia(img)) {
                        return img;
                    }
                }


                node = node.parentElement;
            }
        }

        return null;
    }

    function controlledXRequest(options) {
        const control = xControl;
        if (getPlatform() !== 'x' || !control) return GM_xmlhttpRequest(options);
        checkCancelled();
        let finished = false;
        let handle;
        const clear = () => {
            if (finished) return false;
            finished = true;
            if (control.abortCurrent === abort) control.abortCurrent = null;
            return true;
        };
        const abort = error => {
            if (finished) return;
            if (typeof handle?.abort !== 'function') throw new Error('下載管理器未提供 abort');
            // Suppress late callbacks before abort, then reject the awaiting operation.
            finished = true;
            try { handle.abort(); } catch (failure) { finished = false; throw failure; }
            if (control.abortCurrent === abort) control.abortCurrent = null;
            options.onerror(error);
        };
        const wrapped = { ...options };
        for (const name of ['onload', 'onerror', 'ontimeout', 'onabort']) {
            wrapped[name] = value => {
                if (!clear()) return;
                if (name === 'onabort') options.onerror(controlError());
                else options[name]?.(value);
            };
        }
        handle = GM_xmlhttpRequest(wrapped);
        if (!finished) control.abortCurrent = abort;
        return handle;
    }

    function requestText(url) {
        return new Promise((resolve, reject) => {
            controlledXRequest({
                method: 'GET',
                url,
                timeout: 30000,

                onload(response) {
                    if (
                        response.status >= 200 &&
                        response.status < 400
                    ) {
                        resolve(response.responseText);
                    } else {
                        reject(
                            new Error(
                                `HTTP ${response.status}`
                            )
                        );
                    }
                },

                onerror: reject,

                ontimeout() {
                    reject(
                        new Error('網路逾時')
                    );
                }
            });
        });
    }


    function requestBuffer(url) {
        return new Promise((resolve, reject) => {
            controlledXRequest({
                method: 'GET',
                url,
                responseType: 'arraybuffer',
                timeout: 45000,

                onload(response) {
                    if (
                        response.status >= 200 &&
                        response.status < 400
                    ) {
                        resolve(response.response);
                    } else {
                        reject(
                            new Error(
                                `HTTP ${response.status}`
                            )
                        );
                    }
                },

                onerror: reject,

                ontimeout() {
                    reject(
                        new Error(
                            '下載片段逾時'
                        )
                    );
                }
            });
        });
    }


    function requestBlobURL(url, mimeType) {
        return new Promise((resolve, reject) => {
            GM_xmlhttpRequest({
                method: 'GET',
                url,
                responseType: 'arraybuffer',
                timeout: 60000,

                onload(response) {
                    if (
                        response.status < 200 ||
                        response.status >= 400
                    ) {
                        reject(
                            new Error(
                                `HTTP ${response.status}`
                            )
                        );

                        return;
                    }

                    const blob =
                        new Blob(
                            [response.response],
                            {
                                type: mimeType
                            }
                        );

                    resolve(
                        URL.createObjectURL(blob)
                    );
                },

                onerror: reject,

                ontimeout() {
                    reject(
                        new Error(
                            '載入 FFmpeg 元件逾時'
                        )
                    );
                }
            });
        });
    }

    function getBestImageURL(img) {
        const candidates = [];

        const srcset =
            img.getAttribute('srcset');

        if (srcset) {
            const entries =
                srcset
                    .split(',')
                    .map(item => {
                        const parts =
                            item.trim().split(/\s+/);

                        let score =
                            parseFloat(parts[1]) || 0;

                        if (
                            parts[1]?.endsWith('x')
                        ) {
                            score *= 1000;
                        }

                        return {
                            url:
                                absoluteURL(parts[0]),

                            score
                        };
                    })
                    .sort(
                        (a, b) =>
                            b.score - a.score
                    );

            candidates.push(
                ...entries.map(
                    item => item.url
                )
            );
        }


        candidates.push(
            absoluteURL(img.currentSrc),
            absoluteURL(img.src)
        );


        if (getPlatform() === 'x') {
            const originals = [];

            for (const candidate of candidates) {
                try {
                    const u =
                        new URL(candidate);

                    if (
                        u.hostname.includes(
                            'pbs.twimg.com'
                        )
                    ) {
                        u.searchParams.set(
                            'name',
                            'orig'
                        );

                        originals.push(
                            u.href
                        );
                    }

                } catch (_) {}
            }

            candidates.unshift(
                ...originals
            );
        }


        return (
            unique(candidates)
                .find(
                    url =>
                        /^https?:\/\//i
                            .test(url)
                ) ||
            ''
        );
    }


    function imageExtension(url) {
        try {
            const u =
                new URL(url);

            const format =
                u.searchParams
                    .get('format')
                    ?.toLowerCase();

            if (
                [
                    'jpg',
                    'jpeg',
                    'png',
                    'webp',
                    'gif'
                ].includes(format)
            ) {
                return (
                    '.' +
                    format.replace(
                        'jpeg',
                        'jpg'
                    )
                );
            }

            const match =
                u.pathname.match(
                    /\.(jpg|jpeg|png|webp|gif)/i
                );

            if (match) {
                return (
                    '.' +
                    match[1]
                        .toLowerCase()
                        .replace(
                            'jpeg',
                            'jpg'
                        )
                );
            }

        } catch (_) {}

        return '.jpg';
    }

    function extractMetaVideoURLs(text) {
        const found = new Set();

        const patterns = [
            /"video_url"\s*:\s*"([^"]+)"/gi,
            /"playable_url"\s*:\s*"([^"]+)"/gi,
            /"playable_url_quality_hd"\s*:\s*"([^"]+)"/gi,
            /"contentUrl"\s*:\s*"([^"]+\.mp4[^"]*)"/gi,
            /"content_url"\s*:\s*"([^"]+\.mp4[^"]*)"/gi
        ];

        for (const pattern of patterns) {
            let match;

            while (
                (
                    match =
                        pattern.exec(text)
                ) !== null
            ) {
                const url =
                    decodeURL(match[1]);

                if (
                    /^https?:\/\//i
                        .test(url)
                ) {
                    found.add(url);
                }
            }
        }


        const blockPattern =
            /"video_versions"\s*:\s*\[([\s\S]*?)\]/gi;

        let block;

        while (
            (
                block =
                    blockPattern.exec(text)
            ) !== null
        ) {
            const urlPattern =
                /"url"\s*:\s*"([^"]+)"/gi;

            let match;

            while (
                (
                    match =
                        urlPattern.exec(block[1])
                ) !== null
            ) {
                const url =
                    decodeURL(match[1]);

                if (
                    /^https?:\/\//i
                        .test(url)
                ) {
                    found.add(url);
                }
            }
        }

        return [...found];
    }


    // Read JSON already delivered to this page, matched by exact post shortcode.
    // Never choose an unrelated recommendation or reply from the full document.
    function getInlineMetaVideoURL(postURL, video) {
        const code = postURL.match(/\/(?:post|p|reels?|tv)\/([^/?#]+)/)?.[1];
        if (!code) return '';
        const records = [];
        const seen = new WeakSet();
        const walk = node => {
            if (!node || typeof node !== 'object' || seen.has(node)) return;
            seen.add(node);
            if (String(node.code || node.shortcode || '') === code) {
                const media = Array.isArray(node.carousel_media) ? node.carousel_media : [node];
                for (const item of media) {
                    const urls = (item.video_versions || []).map(v => v.url)
                        .concat(item.video_url || [], item.playable_url || [])
                        .filter(url => typeof url === 'string' && /^https?:\/\//i.test(url));
                    if (urls.length) records.push({ urls, posters: (item.image_versions2?.candidates || []).map(p => p.url) });
                }
            }
            for (const value of Object.values(node)) walk(value);
        };
        for (const script of document.querySelectorAll('script[type="application/json"]')) {
            const text = script.textContent || '';
            if (!text.includes(code)) continue;
            try { walk(JSON.parse(text)); } catch (_) {}
        }
        const key = url => {
            try { return new URL(url).pathname; } catch (_) { return ''; }
        };
        const poster = key(video.poster || video.getAttribute('poster') || '');
        const matched = poster ? records.filter(r => r.posters.some(p => key(p) === poster)) : [];
        const urls = unique((matched.length ? matched : records).flatMap(r => r.urls));
        // Multiple carousel videos require a poster match; otherwise use the existing resolver.
        if (!matched.length && records.length > 1 && new Set(records.map(r => r.urls[0])).size > 1) return '';
        return urls.sort((a,b) => scoreMeta(b) - scoreMeta(a))[0] || '';
    }

    function scoreMeta(url) {
        let score = 0;

        if (/\.mp4/i.test(url)) {
            score += 1000;
        }

        if (/hd/i.test(url)) {
            score += 100;
        }

        return score;
    }

    function findThreadsPostURL(media) {
        const canonical=href=>{
            const match=String(href||'').match(/\/@([^/]+)\/post\/([^/?#]+)/);
            return match ? location.origin+'/@'+match[1]+'/post/'+match[2] : '';
        };
        for(let node=media,i=0;node && i<22;node=node.parentElement,i++) {
            if(node.querySelectorAll?.('video').length>1) break;
            const urls=[...new Set([...node.querySelectorAll?.('a[href*="/post/"]')||[]].map(link=>canonical(link.getAttribute('href'))).filter(Boolean))];
            if(urls.length===1) return urls[0];
            if(urls.length>1) break;
            if(node.matches?.('article,[role="article"]')) break;
        }
        return canonical(location.pathname);
    }

    async function resolveThreadsVideoCore(video) {
        scanInstagramScripts();
        const targetURL=findThreadsPostURL(video);
        const targetCode=findInstagramLocalCode(video)||targetURL.match(/\/post\/([^/?#]+)/)?.[1]||'';
        let cached=selectInstagramCached(video,targetCode);
        if(cached) return {...cached,filename:cached.filename.replace('Instagram_','Threads_')};
        const directSource = getDirectVideoSrc(video);
        if (directSource) {
            return { url: directSource, filename: 'Threads_' + Date.now() + '.mp4' };
        }
        for(let attempt=0;attempt<4;attempt++) {
            await new Promise(resolve=>setTimeout(resolve,200));
            cached=selectInstagramCached(video,targetCode);
            if(cached) return {...cached,filename:cached.filename.replace('Instagram_','Threads_')};
        }
        const postURL = targetURL;

        const match =
            postURL.match(
                /\/@([^/]+)\/post\/([^/?#]+)/
            );

        if (!match) {
            throw new Error(
                '找不到 Threads 貼文'
            );
        }

        const username =
            match[1];

        const code =
            match[2];

        const inlineSource = getInlineMetaVideoURL(postURL, video);
        if (inlineSource) {
            return { url: inlineSource, filename: 'Threads_' + username + '_' + code + '.mp4' };
        }
        const html =
            await requestText(postURL);

        const found =
            new Set();

        let pos = 0;

        while (true) {
            const index =
                html.indexOf(
                    code,
                    pos
                );

            if (index < 0) {
                break;
            }

            const block =
                html.slice(
                    Math.max(
                        0,
                        index - 50000
                    ),

                    Math.min(
                        html.length,
                        index + 120000
                    )
                );

            if (
                /video_versions|video_url|playable_url/
                    .test(block)
            ) {
                extractMetaVideoURLs(block)
                    .forEach(
                        url =>
                            found.add(url)
                    );
            }

            pos =
                index +
                code.length;
        }


        const urls =
            [...found]
                .filter(
                    url =>
                        /\.mp4|fbcdn|cdninstagram/i
                            .test(url)
                )
                .sort(
                    (a, b) =>
                        scoreMeta(b) -
                        scoreMeta(a)
                );

        if (urls.length) {
            return {
                url:
                    urls[0],

                filename:
                    `Threads_${username}_${code}.mp4`
            };
        }


        const direct =
            getDirectVideoSrc(video);

        if (direct) {
            return {
                url:
                    direct,

                filename:
                    `Threads_${username}_${code}.mp4`
            };
        }


        throw new Error(
            '找不到 Threads 影片來源'
        );
    }

    function findInstagramLocalCode(media) {
        const checkedFibers=new WeakSet();
        for(let element=media,depth=0;element && depth<16;element=element.parentElement,depth++) {
            if(element.querySelectorAll?.('video').length>1) break;
            const codes=new Set(),seen=new WeakSet();let visits=0;
            const add=code=>{if(typeof code==='string' && /^[A-Za-z0-9_-]{1,28}$/.test(code)) codes.add(code);};
            function walk(value) {
                if(!value || typeof value!=='object' || seen.has(value) || ++visits>5000) return;
                seen.add(value);
                if(Array.isArray(value.video_versions) || typeof value.video_url==='string' || value.media_type===2 || value.is_video===true || /Video/.test(value.__typename||'')) {
                    inspectInstagramMedia(value);
                    add(value.code||value.shortcode||metaCodeFromId(value.pk||value.id));
                }
                for(const key of ['videoId','mediaId','media_id','video_id']) add(metaCodeFromId(value[key]));
                for(const [key,child] of Object.entries(value)) {
                    if(['return','stateNode','alternate','_owner','ownerDocument','parentElement','child','sibling'].includes(key)) continue;
                    walk(child);
                }
            }
            for(const key of Object.keys(element)) {
                if(key.startsWith('__reactProps$')) walk(element[key]);
                else if(key.startsWith('__reactFiber$')) {
                    // React component parents may not have their own DOM element.
                    let fiber=element[key];
                    for(let level=0;fiber && level<20;level++,fiber=fiber.return) {
                        if(checkedFibers.has(fiber)) break;
                        checkedFibers.add(fiber);walk(fiber.memoizedProps);walk(fiber.pendingProps);
                        if(codes.size>1 || codes.size===1) break;
                        if(fiber.stateNode?.querySelectorAll?.('video').length>1) break;
                    }
                }
            }
            const imageKeys=[media.poster||'',...(element.querySelectorAll?.('img')||[])].map(value=>igImageKey(typeof value==='string'?value:value.currentSrc||value.src||'')).filter(key=>key && key!=='/');
            const matching=[...igMediaCache.values()].filter(record=>record.images.some(key=>imageKeys.includes(key)));
            const matched=[...new Set(matching.map(record=>record.code))];
            if(matched.length===1 && (!codes.size || codes.has(matched[0]))) return matched[0];
            if(codes.size===1) return [...codes][0];
            if(codes.size>1) break;
            if(element.matches?.('article,[role="article"]')) break;
        }
        return '';
    }

    function findInstagramPostURL(media) {
        const localCode=findInstagramLocalCode(media);
        if(localCode) return location.origin+'/p/'+localCode+'/';
        const canonical=href=>{
            const match=String(href||'').match(/\/(?:[^/?#]+\/)?(p|reels?|tv)\/([^/?#]+)/);
            return match ? location.origin+'/'+(match[1]==='reels'?'reel':match[1])+'/'+match[2]+'/' : '';
        };
        // The nearest single-video container takes precedence over the SPA route.
        for(let node=media,i=0;node && i<20;node=node.parentElement,i++) {
            if(node.querySelectorAll?.('video').length>1) break;
            const urls=[...new Set([...node.querySelectorAll?.('a[href]')||[]].map(link=>canonical(link.getAttribute('href'))).filter(Boolean))];
            if(urls.length===1) return urls[0];
            if(urls.length>1) break;
            if(node.matches?.('article,[role="article"]')) break;
        }
        return canonical(location.pathname);
    }

            if(cached) return cached;
        }
        const direct=getDirectVideoSrc(video);

        cached=selectInstagramCached(video,code);

        const html=await requestText(postURL);

    function getXMediaId(video) {
        const values = [
            video.poster,
            video.getAttribute('poster'),
            video.currentSrc,
            video.src
        ];

        const patterns = [
            /amplify_video_thumb\/(\d+)/i,
            /ext_tw_video_thumb\/(\d+)/i,
            /tweet_video_thumb\/(\d+)/i,
            /amplify_video\/(\d+)/i,
            /ext_tw_video\/(\d+)/i
        ];

        for (const value of values) {
            if (!value) continue;

            for (const pattern of patterns) {
                const match =
                    value.match(pattern);

                if (match) {
                    return match[1];
                }
            }
        }

        return '';
    }

    function parseAttributeList(text) {
        const result = {};

        const regex =
            /([A-Z0-9-]+)=("[^"]*"|[^,]*)/gi;

        let match;

        while (
            (
                match =
                    regex.exec(text)
            ) !== null
        ) {
            let value =
                match[2];

            if (
                value.startsWith('"') &&
                value.endsWith('"')
            ) {
                value =
                    value.slice(1, -1);
            }

            result[
                match[1].toUpperCase()
            ] = value;
        }

        return result;
    }


    function parseMasterPlaylist(
        text,
        masterURL
    ) {
        const lines =
            text
                .split(/\r?\n/)
                .map(
                    line => line.trim()
                )
                .filter(Boolean);

        const audio = [];
        const video = [];


        for (
            let i = 0;
            i < lines.length;
            i++
        ) {
            const line =
                lines[i];


            if (
                line.startsWith(
                    '#EXT-X-MEDIA:'
                )
            ) {
                const attrs =
                    parseAttributeList(
                        line.slice(
                            '#EXT-X-MEDIA:'
                                .length
                        )
                    );

                if (
                    attrs.TYPE ===
                    'AUDIO' &&
                    attrs.URI
                ) {
                    audio.push({
                        ...attrs,

                        url:
                            absoluteURL(
                                attrs.URI,
                                masterURL
                            )
                    });
                }
            }


            if (
                line.startsWith(
                    '#EXT-X-STREAM-INF:'
                )
            ) {
                const attrs =
                    parseAttributeList(
                        line.slice(
                            '#EXT-X-STREAM-INF:'
                                .length
                        )
                    );

                let j =
                    i + 1;

                while (
                    j < lines.length &&
                    lines[j]
                        .startsWith('#')
                ) {
                    j++;
                }

                if (
                    j <
                    lines.length
                ) {
                    video.push({
                        ...attrs,

                        url:
                            absoluteURL(
                                lines[j],
                                masterURL
                            )
                    });
                }
            }
        }


        return {
            audio,
            video
        };
    }


    function findKnownXChildPlaylists(
        mediaId
    ) {
        const videos = [];
        const audios = [];

        for (
            const url

                )
            ) {
                continue;
            }

            if (
                !/\.m3u8(?:\?|$)/i

            ) {
                continue;
            }

            if (
                /\/pl\/avc1\//i
                    .test(url)
            ) {
                videos.push(url);
            }

            if (
                /\/pl\/mp4a\//i
                    .test(url)
            ) {
                audios.push(url);
            }
        }


        return {
            videos:
                unique(videos),

            audios:
                unique(audios)
        };
    }

    function resolutionScore(url) {
        const match =
            url.match(
                /(\d+)x(\d+)/
            );

        if (!match) {
            return 0;
        }

        return (
            Number(match[1]) *
            Number(match[2])
        );
    }


    function audioScore(url) {
        const match =
            url.match(
                /\/mp4a\/(\d+)/
            );

        return (
            Number(
                match?.[1]
            ) ||
            0
        );
    }

    function parseMediaPlaylist(
        text,
        playlistURL
    ) {
        const lines =
            text
                .split(/\r?\n/)
                .map(
                    line => line.trim()
                )
                .filter(Boolean);

        let initURL = '';

        const segments = [];


        for (const line of lines) {
            if (
                line.startsWith(
                    '#EXT-X-MAP:'
                )
            ) {
                const attrs =
                    parseAttributeList(
                        line.slice(
                            '#EXT-X-MAP:'
                                .length
                        )
                    );

                if (
                    attrs.URI
                ) {
                    initURL =
                        absoluteURL(
                            attrs.URI,
                            playlistURL
                        );
                }

                continue;
            }


            if (
                line.startsWith('#')
            ) {
                continue;
            }


            segments.push(
                absoluteURL(
                    line,
                    playlistURL
                )
            );
        }


        return {
            initURL,

            segments:
                unique(segments)
        };
    }

    async function downloadTrack(
        playlistURL,
        label
    ) {
        progress(
            `X：讀取${label}播放清單…`
        );

        const playlistText =
            await requestText(
                playlistURL
            );

        const playlist =
            parseMediaPlaylist(
                playlistText,
                playlistURL
            );

        if (
            !playlist.initURL
        ) {
            throw new Error(
                `${label}播放清單沒有 EXT-X-MAP`
            );
        }

        if (
            !playlist.segments.length
        ) {
            throw new Error(
                `${label}播放清單沒有媒體片段`
            );
        }


        const buffers = [];

        progress(
            `X：下載${label}初始化資料…`
        );

        buffers.push(
            await requestBuffer(
                playlist.initURL
            )
        );


        for (
            let i = 0;
            i < playlist.segments.length;
            i++
        ) {
            progress(
                `X：下載${label} ${Math.round(i / playlist.segments.length * 100)}%｜${i}/${playlist.segments.length}`
            );

            buffers.push(
                await requestBuffer(
                    playlist.segments[i]
                )
            );
        }


        let total = 0;

        for (
            const buffer
            of buffers
        ) {
            total +=
                buffer.byteLength;
        }


        const output =
            new Uint8Array(total);

        let offset = 0;

        for (
            const buffer
            of buffers
        ) {
            const bytes =
                new Uint8Array(buffer);

            output.set(
                bytes,
                offset
            );

            offset +=
                bytes.byteLength;
        }


        checkCancelled();
        progress(`X：${label} 100%｜${formatBytes(output.byteLength)}`);
        return output;
    }

    async function ensureFFmpegWorkerSource() {
        if (ffmpegWorkerSource) {
            return ffmpegWorkerSource;
        }

        if (ffmpegWorkerLoadingPromise) {
            return ffmpegWorkerLoadingPromise;
        }

        ffmpegWorkerLoadingPromise = (async () => {
            progress('X：載入 MP4 合併器（約 10 MB）…');

            const source = await requestText(FFMPEG_WORKER_URL);

            if (!source || source.length < 1000000) {
                throw new Error('FFmpeg Worker 下載內容異常');
            }

            ffmpegWorkerSource = source;

            console.log(
                '[SMD:X] FFmpeg Worker loaded:',
                (source.length / 1024 / 1024).toFixed(2),
                'MB'
            );

            return source;
        })();

        try {
            return await ffmpegWorkerLoadingPromise;
        } catch (error) {
            ffmpegWorkerLoadingPromise = null;
            throw error;
        }
    }


    function runFFmpegWorker(videoBytes, audioBytes) {
        return new Promise(async (resolve, reject) => {
            let worker = null;
            let workerURL = '';
            let settled = false;
            let timeout;
            const control = xControl;
            const abortWorker = error => fail(error);

            const cleanup = () => {
                clearTimeout(timeout);
                if (control?.abortCurrent === abortWorker) control.abortCurrent = null;
                try {
                    worker?.terminate();
                } catch (_) {}

                if (workerURL) {
                    URL.revokeObjectURL(workerURL);
                }
            };

            const fail = error => {
                if (settled) return;
                settled = true;
                cleanup();
                reject(error instanceof Error ? error : new Error(String(error)));
            };

            try {
                const source = await ensureFFmpegWorkerSource();
                checkCancelled();
                if (control) control.abortCurrent = abortWorker;

                progress('X：啟動 MP4 合併器…');

                const workerBlob = new Blob(
                    [source],
                    { type: 'text/javascript' }
                );

                workerURL = URL.createObjectURL(workerBlob);
                worker = new Worker(workerURL);

                timeout = setTimeout(() => {
                    fail(new Error('FFmpeg Worker 合併逾時'));
                }, 180000);

                worker.onerror = event => {
                    clearTimeout(timeout);
                    fail(
                        new Error(
                            `FFmpeg Worker 啟動失敗：${event.message || 'unknown error'}`
                        )
                    );
                };

                worker.onmessage = event => {
                    const msg = event.data || {};

                    if (msg.type === 'ready') {
                        progress('X：FFmpeg 已就緒，正在合併影片＋聲音…');

                        // ffmpeg.js worker 的 MEMFS 介面。
                        // 兩個輸入都是我們前面已經組好的完整 fragmented MP4 track。
                        worker.postMessage({
                            type: 'run',
                            MEMFS: [
                                {
                                    name: 'video.mp4',
                                    data: videoBytes
                                },
                                {
                                    name: 'audio.m4a',
                                    data: audioBytes
                                }
                            ],
                            arguments: [
                                '-i', 'video.mp4',
                                '-i', 'audio.m4a',
                                '-map', '0:v:0',
                                '-map', '1:a:0',
                                '-c:v', 'copy',
                                '-c:a', 'copy',
                                '-shortest',
                                '-movflags', '+faststart',
                                'output.mp4'
                            ]
                        });

                        return;
                    }

                    if (msg.type === 'stdout' || msg.type === 'stderr') {
                        console.log(`[SMD:FFmpeg:${msg.type}]`, msg.data);
                        return;
                    }

                    if (msg.type === 'exit') {
                        console.log('[SMD:FFmpeg] exit:', msg.data);
                        return;
                    }

                    if (msg.type === 'done') {
                        clearTimeout(timeout);

                        try {
                            const files = msg.data?.MEMFS || [];
                            const output = files.find(file => file.name === 'output.mp4');

                            if (!output?.data?.length) {
                                throw new Error('FFmpeg 完成但找不到 output.mp4');
                            }

                            const result = new Uint8Array(output.data.length);
                            result.set(output.data);

                            if (settled) return;
                            settled = true;
                            cleanup();
                            resolve(result);
                        } catch (error) {
                            fail(error);
                        }
                    }
                };

            } catch (error) {
                fail(error);
            }
        });
    }

        }

        // ----------------------------------------------------
        // ENGINE A：X API MP4 DIRECT
        // ----------------------------------------------------

        const apiRecord = xApiMediaCache.get(String(mediaId));
        const directMP4 = apiRecord?.best?.url || '';

        if (directMP4) {
            const bitrate = apiRecord.best.bitrate || 0;
            const base = cleanFilename(

            progress(
                `X：⚡ API MP4 直載｜${bitrate ? Math.round(bitrate / 1000) + ' kbps' : '最高可用畫質'}`

                '[SMD:X] Engine A = API MP4 direct',
                { mediaId, tweetId: info.tweetId, bitrate, url: directMP4 }
            );

            xControl.phase = 'mp4';
            try {
                await new Promise((resolve, reject) => {
                    const control = xControl;
                    let handle;
                    let finished = false;
                    let loaded = 0, total = 0;
                    let sampleLoaded = 0, sampleAt = performance.now();
                    let speed = 0, lastChangeAt = Date.now();
                    const render = () => {
                        if (finished) return;
                        const idle = Date.now() - lastChangeAt;
                        const pct = total > 0 ? Math.min(100, Math.round(loaded / total * 100)) + '%' : '總大小未知';
                        const rate = idle >= 10000 ? 0 : speed;
                        const speedText = rate >= 1024 * 1024
                            ? (rate / 1024 / 1024).toFixed(2) + ' MB/s'
                            : (rate / 1024).toFixed(1) + ' KB/s';
                        progress(`X：⚡ MP4 直載｜${pct}｜${formatBytes(loaded)}${total > 0 ? ' / ' + formatBytes(total) : ''}｜${speedText}${idle >= 10000 ? '｜等待資料中…' : ''}`);
                    };
                    const watchdog = setInterval(render, 1000);
                    const finish = (error) => {
                        if (finished) return;
                        finished = true;
                        clearInterval(watchdog);
                        if (control.abortCurrent === abort) control.abortCurrent = null;
                        error ? reject(error) : resolve();
                    };
                    const abort = error => {
                        if (finished) return;
                        if (typeof handle?.abort !== 'function') throw new Error('GM_download 未提供 abort，無法安全切換 HLS');
                        // abort is called before this promise rejects and Engine B can start.
                        finished = true;
                        try { handle.abort(); } catch (failure) { finished = false; throw failure; }
                        finished = false;
                        finish(error);
                    };
                    try {
                        handle = GM_download({
                            url: directMP4, name: base + '.mp4', saveAs: true,
                            onprogress: event => {
                                if (finished) return;
                                loaded = Number(event?.loaded) || 0;
                                total = Number(event?.total) || 0;
                                const now = performance.now();
                                if (loaded !== sampleLoaded) lastChangeAt = Date.now();
                                if (now - sampleAt >= 500) {
                                    speed = Math.max(0, loaded - sampleLoaded) * 1000 / (now - sampleAt);
                                    sampleLoaded = loaded;
                                    sampleAt = now;
                                }
                                render();
                            },
                            onload: () => finish(),
                            onerror: error => finish(new Error('X：API MP4 直載失敗：' + (error?.error || 'download error'))),
                            ontimeout: () => finish(new Error('X：API MP4 直載逾時')),
                            onabort: () => finish(controlError())
                        });
                        if (!finished) control.abortCurrent = abort;
                        render();
                    } catch (error) { finish(error); }
                });
                checkCancelled();

            } catch (error) {

            directMP4 ? 'X：MP4 已中止，啟動 HLS 引擎…' : 'X：API 沒有命中 MP4，切換 HLS 引擎…'

                'X：找不到影片播放清單'

    let fbDiagnostic = null;

    function inspectFacebookData(doc, mediaId) {
        const stats = { jsonScripts: 0, targetScripts: 0, parsedScripts: 0, parseFailures: 0, matchedRecords: 0, nativeFields: 0, nativeURLs: 0, matchedKeys: [], transportShapes: [] };
        const keys = new Set();
        const shapes = new Set();
        const inspectShape = (value, path = '', depth = 0) => {
            if (depth > 7 || !value || typeof value !== 'object') return;
            for (const [key, child] of Object.entries(value)) {
                if (Array.isArray(value) && Number(key) > 2) break;
                const next = path ? path + '.' + key : key;
                if (/delivery|dash|manifest|progressive|playable|representation|stream|url|mime|codec|video|audio/i.test(key)) {
                    const kind = child === null ? 'null' : Array.isArray(child) ? 'array[' + child.length + ']' : typeof child;
                    const marker = typeof child === 'string' && /^https?:\/\//i.test(child) ? ':https-url'
                        : typeof child === 'string' && /<MPD|<Period|<Representation/.test(child) ? ':mpd-xml' : '';
                    shapes.add(next + ':' + kind + marker);
                }
                if (child && typeof child === 'object') inspectShape(child,next,depth+1);
            }
        };
        for (const script of doc.querySelectorAll('script[type="application/json"]')) {
            stats.jsonScripts++;
            const text = script.textContent || '';
            if (!text.includes(mediaId)) continue;
            stats.targetScripts++;
            const visit = node => {
                if (!node || typeof node !== 'object') return;
                if (String(node.id || node.video_id || '') === mediaId) {
                    stats.matchedRecords++;
                    for (const key of Object.keys(node)) keys.add(key);
                    // Inspect only the matching video's transport subtrees.
                    for (const [key, value] of Object.entries(node)) {
                        if (/delivery|dash|manifest|progressive|playable|representation|stream|url|video_data|audio/i.test(key)) {
                            const kind = value === null ? 'null' : Array.isArray(value) ? 'array[' + value.length + ']' : typeof value;
                            shapes.add(key + ':' + kind + (typeof value === 'string' && /<MPD/.test(value) ? ':mpd-xml' : ''));
                            inspectShape(value,key);
                        }
                    }
                    const fields = node.videoDeliveryLegacyFields || node;
                    for (const key of ['browser_native_hd_url','browser_native_sd_url','playable_url_quality_hd','playable_url']) {
                        if (Object.prototype.hasOwnProperty.call(fields, key)) stats.nativeFields++;
                        if (typeof fields[key] === 'string' && /^https?:\/\//i.test(fields[key])) stats.nativeURLs++;
                    }
                }
                for (const value of Object.values(node)) visit(value);
            };
            try { visit(JSON.parse(text)); stats.parsedScripts++; } catch (_) { stats.parseFailures++; }
        }
        stats.matchedKeys = [...keys].sort();
        stats.transportShapes = [...shapes].sort().slice(0,240);
        return stats;
    }

    function facebookVideoId(url) {
        try {
            const parsed = new URL(url, location.href);
            if (!(parsed.hostname === 'facebook.com' || parsed.hostname.endsWith('.facebook.com'))) return '';
            return parsed.pathname.match(/\/(?:reel|videos)\/(?:[^/]+\/)?(\d+)(?:\/|$)/)?.[1]
                || parsed.searchParams.get('v')?.match(/^\d+$/)?.[0] || '';
        } catch (_) { return ''; }
    }

    function facebookQuality(candidate) {
        let width=Number(candidate.width)||0, height=Number(candidate.height)||0, bitrate=0;
        try {
            const url=new URL(candidate.url);
            bitrate=Number(url.searchParams.get('bitrate'))||0;
            const encoded=url.searchParams.get('efg');
            if(encoded) {
                const meta=JSON.parse(atob(encoded.replace(/-/g,'+').replace(/_/g,'/')));
                const tag=String(meta.vencode_tag || '');
                const p=tag.match(/(?:^|[_.])(\d{3,4})p(?:[_.]|$)/)?.[1];
                if(!height && p)height=Number(p);
                bitrate=bitrate || Number(meta.bitrate)||0;
            }
        }catch(_){}
        return {height,width,bitrate,score:Number(candidate.score)||0};
    }

    function chooseFacebookSource(candidates) {
        return [...candidates].filter(c=>c?.url).sort((a,b)=>{
            const x=facebookQuality(a), y=facebookQuality(b);
            if(x.height && y.height) return y.height-x.height || y.width-x.width || y.score-x.score || y.bitrate-x.bitrate;
            return y.score-x.score || y.height-x.height || y.width-x.width || y.bitrate-x.bitrate;
        })[0]?.url || '';
    }

    function facebookThumbnailKey(url) {
        try {
            const parsed = new URL(url, location.href);
            if (!/^https?:$/.test(parsed.protocol)) return '';
            // Image resizing and CDN signatures change query parameters.
            return parsed.pathname;
        } catch (_) { return ''; }
    }

    function matchFacebookThumbnail(urls) {
        const keys = new Set(urls.map(facebookThumbnailKey).filter(Boolean));
        if (!keys.size) return '';
        const matches = [];
        for (const [id, record] of fbApiMediaCache) {
            if ((record.thumbnails || []).some(url => keys.has(facebookThumbnailKey(url)))) matches.push(id);
        }
        return matches.length === 1 ? matches[0] : '';
    }

    function findFacebookVideoId(video) {
        const posterId = matchFacebookThumbnail([video.poster || video.getAttribute('poster') || '']);
        if (posterId) return posterId;
        let node = video;
        for (let depth = 0; depth < 18 && node && node !== document.body && node !== document.documentElement; depth++, node = node.parentElement) {
            // Never search a container holding another player or multiple posts.
            if (node.querySelectorAll('video').length > 1 || node.querySelectorAll('[role="article"]').length > 1) break;
            const explicit = [node.getAttribute('data-video-id'),node.getAttribute('data-videoid')]
                .filter(value => /^\d+$/.test(value || ''));
            const ids = unique([...explicit,...[...node.querySelectorAll('a[href]')].map(a => facebookVideoId(a.href))]);
            if (ids.length === 1) return ids[0];
            if (ids.length > 1) break;
            const thumbnailId = matchFacebookThumbnail([...node.querySelectorAll('img')].map(img => img.currentSrc || img.src));
            if (thumbnailId) return thumbnailId;
            // The hovered post is the boundary; no links from sibling recommendations.
            if (node.getAttribute('role') === 'article') break;
        }
        const visible = [...document.querySelectorAll('video')].filter(v => {
            const rect = v.getBoundingClientRect();
            return rect.width > 0 && rect.height > 0 && rect.bottom > 0 && rect.top < innerHeight
                && rect.right > 0 && rect.left < innerWidth;
        });
        return visible.length === 1 && visible[0] === video ? facebookVideoId(location.href) : '';
    }

    function extractFacebookNativeURL(root, mediaId) {
        const candidates = [];
        const seen = new WeakSet();
        const walk = node => {
            if (!node || typeof node !== 'object' || seen.has(node)) return;
            seen.add(node);
            const id = String(node.id || node.video_id || '');
            if (id === mediaId) {
                const fields = node.videoDeliveryLegacyFields || node;
                for (const [key, score] of [
                    ['browser_native_hd_url', 4], ['playable_url_quality_hd', 3],
                    ['browser_native_sd_url', 2], ['playable_url', 1]
                ]) {
                    const url = fields[key];
                    if (typeof url === 'string' && /^https?:\/\//i.test(url)) candidates.push({url, score});
                }
            }
            for (const value of Object.values(node)) walk(value);
        };
        walk(root);
        return candidates.sort((a,b) => b.score - a.score)[0]?.url || '';
    }

    function findFacebookNativeURL(doc, mediaId) {
        for (const script of doc.querySelectorAll('script[type="application/json"]')) {
            const text = script.textContent || '';
            if (!text.includes(mediaId) || !/browser_native|playable_url/.test(text)) continue;
            try {
                const url = extractFacebookNativeURL(JSON.parse(text), mediaId);
                if (url) return url;
            } catch (_) {}
        }
        return '';
    }

    function requestFacebookPublicText(mediaId) {
        return new Promise((resolve, reject) => {
            GM_xmlhttpRequest({
                method: 'GET',
                url: 'https://www.facebook.com/reel/' + mediaId + '/',
                // Logged-in FB can omit native MP4 fields. Read the public page
                // without session cookies, matching the verified logged-out case.
                anonymous: true,
                timeout: 30000,
                onload(response) {
                    if (fbDiagnostic) {
                        const text = response.responseText || '';
                        fbDiagnostic.publicResponse = {
                            status: response.status,
                            length: text.length,
                            finalVideoId: facebookVideoId(response.finalUrl || ''),
                            containsTargetId: text.includes(mediaId),
                            containsNativeHD: text.includes('browser_native_hd_url'),
                            containsNativeSD: text.includes('browser_native_sd_url'),
                            containsLoginForm: /name=["'](?:email|pass)["']/.test(text),
                            containsCheckpoint: /checkpoint|captcha/i.test(text)
                        };
                    }
                    if (response.status >= 200 && response.status < 400) resolve(response.responseText);
                    else reject(new Error('FB 公開頁面 HTTP ' + response.status));
                },
                onerror: () => reject(new Error('FB 公開頁面請求失敗')),
                ontimeout: () => reject(new Error('FB 公開頁面請求逾時'))
            });
        });
    }

    async function resolveFacebookVideo(video) {
        const direct = getDirectVideoSrc(video);
        // The playing source may be a low-resolution adaptive choice. Keep it as fallback.

        // Populate thumbnail identities before trying to match a feed player.
        for (const script of document.querySelectorAll('script[type="application/json"]')) inspectFacebookAPI(script.textContent);
        document.getElementById('smd-fb-diagnostic')?.remove();
        const mediaId = findFacebookVideoId(video);
        fbDiagnostic = {
            version: VERSION,
            routeVideoId: facebookVideoId(location.href),
            selectedVideoId: mediaId,
            totalVideoElements: document.querySelectorAll('video').length,
            sourceType: (video.currentSrc || video.src || '').startsWith('blob:') ? 'blob' : 'empty-or-other'
        };
        if (!mediaId) {
            if (direct) return {url:direct,filename:'Facebook_'+Date.now()+'.mp4'};
            throw new Error('FB：請先點開這支影片，再按下載');
        }
        // Read any remaining bootstrap scripts before consulting the API cache.
        for (const script of document.querySelectorAll('script[type="application/json"]')) inspectFacebookAPI(script.textContent);
        const cached = fbApiMediaCache.get(mediaId);
        fbDiagnostic.apiCache = {
            records: fbApiMediaCache.size,
            targetFound: !!cached,
            progressiveURLs: cached?.urls?.length || 0,
            hasDashManifest: !!cached?.manifest,
            transportShapes: cached?.shapes || []
        };
        const cachedURL = chooseFacebookSource(cached?.urls || []);
        fbDiagnostic.currentPage = inspectFacebookData(document, mediaId);
        const pageURL = findFacebookNativeURL(document, mediaId);
        let url = chooseFacebookSource([...(cached?.urls || []),...(pageURL ? [{url:pageURL,score:4}] : [])]);
        if (!url && direct) url=direct;
        if (!url) {
            // SPA navigation may not leave the new Reel's JSON in the current DOM.
            try {
                const html = await requestFacebookPublicText(mediaId);
                const doc = new DOMParser().parseFromString(html, 'text/html');
                fbDiagnostic.publicPage = inspectFacebookData(doc, mediaId);
                url = findFacebookNativeURL(doc, mediaId);
            } catch (error) {
                fbDiagnostic.requestError = error.message || 'unknown';
                showFacebookDiagnostic();
                throw error;
            }
        }
        if (!url) {
            showFacebookDiagnostic();
            throw new Error('FB：尚未取得 MP4，請下載左上方的診斷 TXT');
        }
        console.log('[SMD:FB] native MP4 matched', mediaId);
        return { url, filename: 'Facebook_' + mediaId + '.mp4' };
    }

                    '[SMD]',
                    error
                );


                closeProgress();


                toast(
                    error?.message ||
                    '影片下載失敗',
                    7000
                );

            } finally {

                xControl = null;
                closeProgress();
                resolving = false;

                button.disabled = false;

                button.textContent =
                    type === 'video'
                        ? '↓ 下載影片'
                        : '↓ 下載照片';
            }
        },
        true
    );


    // ========================================================
    // SPA
    // ========================================================

    let lastURL =
        location.href;


    setInterval(
        () => {
            if (
                lastURL !==
                location.href
            ) {
                lastURL =
                    location.href;

                activeMedia = null;
                activeType = null;

                if (
                    !button.disabled
                ) {
                    button.style.display =
                        'none';
                }
            }
        },
        700
    );


    console.log(
        `%c Social Media Downloader v${VERSION} | ${platformLabel()} `,
        'background:#111;color:#fff;padding:6px 10px;border-radius:6px;font-weight:bold'
    );
}


// ============================================================
// DOM READY
// ============================================================

if (
    document.readyState ===
    'loading'
) {
    document.addEventListener(
        'DOMContentLoaded',
        start,
        {
            once: true
        }
    );

} else {
    start();
}

})();
